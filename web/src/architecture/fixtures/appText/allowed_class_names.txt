export function Chip({ on, rail }: { on: boolean; rail: string }) {
  return (
    <div className={on ? 'chip chip--on' : 'chip'} data-state="open" data-label={`row ${rail}`}>

      <b className={railClassName(rail, on)} />
    </div>
  )
}
