## How this was taken

Procedure script, kept outside the repository (`~/.cache/apostrophe-GH-247/run.sh`, work directory `~/.cache/apostrophe-GH-247/<head-sha>/`, completion markers per step). Run as `run.sh <checkout>`.

```bash
#!/usr/bin/env bash
# GH-247 evidence procedure. Kept outside the repository; never committed.
# usage: run.sh <repo-checkout>   (head = the checkout's HEAD; work dir keyed by that SHA)
set -euo pipefail
REPO=$(cd "$1" && pwd)
cd "$REPO"
HEAD_SHA=$(git rev-parse HEAD)
GH141_MERGE=$(git rev-parse 0208925)
GH141_BASE=$(git rev-parse 0208925^1)
W=$HOME/.cache/apostrophe-GH-247/$HEAD_SHA
SCRATCH=$W/scratch
mkdir -p "$W"
TZ_AR=America/Argentina/Buenos_Aires
PORT=0
now() { TZ=$TZ_AR date '+%Y-%m-%d %H:%M:%S %z'; }
done_() { [ -e "$W/$1" ]; }

# ---- Step 1: build "before" and prove it (marker: undone-tree)
if ! done_ undone-tree; then
  [ -z "$(git status --porcelain)" ] || { echo "pending changes"; exit 1; }
  rm -rf "$SCRATCH" "$W/tmp-index"
  GIT_INDEX_FILE=$W/tmp-index git read-tree "$HEAD_SHA"
  git diff --binary "$GH141_BASE" "$GH141_MERGE" | GIT_INDEX_FILE=$W/tmp-index git apply -R --cached
  UNDONE=$(GIT_INDEX_FILE=$W/tmp-index git write-tree)
  {
    echo "head=$HEAD_SHA"; echo "undone-tree=$UNDONE"
    git diff-tree -r --name-only "$HEAD_SHA" "$UNDONE" | sort > "$W/paths-undone"
    git diff-tree -r --name-only "$GH141_BASE" "$GH141_MERGE" | sort > "$W/paths-gh141"
    cmp "$W/paths-undone" "$W/paths-gh141" && echo "path sets equal: $(wc -l < "$W/paths-gh141") paths"
    bad=0
    while IFS= read -r p; do
      a=$(git rev-parse -q --verify "$UNDONE:$p" || echo MISSING)
      b=$(git rev-parse -q --verify "$GH141_BASE:$p" || echo MISSING)
      if [ "$a" != "$b" ]; then echo "MISMATCH $p $a $b"; bad=1; fi
    done < "$W/paths-gh141"
    [ $bad = 0 ] && echo "blob ids equal for every path"
    [ $bad = 0 ] || exit 1
  } > "$W/undone-tree.tmp"
  mkdir -p "$SCRATCH"
  git archive "$UNDONE" | tar -x -C "$SCRATCH"
  (cd "$SCRATCH" && find . -type f | sed 's|^\./||' | sort) > "$W/scratch-files"
  git ls-tree -r --name-only "$UNDONE" | sort > "$W/undone-files"
  cmp "$W/scratch-files" "$W/undone-files"
  echo "scratch file list equals ls-tree of undone tree ($(wc -l < "$W/undone-files") files)" >> "$W/undone-tree.tmp"
  ln -s "$REPO/.venv" "$SCRATCH/.venv"
  ln -s "$REPO/web/node_modules" "$SCRATCH/web/node_modules"
  echo "$UNDONE" > "$W/undone-sha"
  mv "$W/undone-tree.tmp" "$W/undone-tree"
fi
UNDONE=$(cat "$W/undone-sha")

# ---- Step 1b: the capture tooling is the same in both trees (marker: tooling.ok)
# Head against the undone tree, not against 0208925^1: later master merges (GH-219) changed
# scripts/design after GH-141, and the scratch tree is the head's with GH-141 undone.
if ! done_ tooling.ok; then
  TOOLING="scripts/design/comparisons.mjs scripts/design/sample_app.py scripts/design/sample-app.sh scripts/design/screens.json scripts/design/screens.mjs scripts/design/fonts.mjs scripts/design/ae.mjs"
  # shellcheck disable=SC2086
  git diff --quiet "$HEAD_SHA" "$UNDONE" -- $TOOLING
  for f in $TOOLING; do cmp "$REPO/$f" "$SCRATCH/$f"; done
  { echo "capture tooling identical at head and in the undone tree (git diff --quiet exit 0; files compared byte for byte in the scratch tree):"; for f in $TOOLING; do echo "  $f"; done; } > "$W/tooling.ok"
fi

# ---- Step 2: smoke test the scratch tree (marker: smoke.ok)
start_server() { # $1 = tree, $2 = out dir, $3 = facts file
  rm -rf "$2"; mkdir -p "$2"
  (cd "$1" && exec setsid scripts/design/sample-app.sh --dir "$2" > "$2/server.out" 2>&1) &
  SERVER_PID=$!
  for _ in $(seq 1 240); do [ -e "$2/sample.json" ] && break; sleep 1; done
  [ -e "$2/sample.json" ] || { echo "sample app did not come up"; cat "$2/server.out"; stop_server; exit 1; }
  cp "$2/sample.json" "$3"
}
stop_server() { pkill -TERM -P "$SERVER_PID" 2>/dev/null || true; kill -TERM "$SERVER_PID" 2>/dev/null || true; wait "$SERVER_PID" 2>/dev/null || true; pkill -f "uvicorn app.main:create_app.*--port $PORT" 2>/dev/null || true; }
if ! done_ smoke.ok; then
  {
    (cd "$SCRATCH" && uv run --no-project python -c 'import sys, os, app; print(os.path.realpath(sys.prefix), app.__file__)') > "$W/smoke-import"
    cat "$W/smoke-import"
    read -r prefix appfile < "$W/smoke-import"
    [ "$prefix" = "$(realpath "$REPO/.venv")" ] || { echo "sys.prefix is not the head's .venv"; exit 1; }
    case "$appfile" in "$SCRATCH"/*) echo "app.__file__ inside scratch";; *) echo "app.__file__ outside scratch"; exit 1;; esac
    start_server "$SCRATCH" "$W/smoke-before" "$W/smoke-facts.json"
    PORT=$(python3 -c "import json;print(json.load(open('$W/smoke-facts.json'))['url'].rsplit(':',1)[1])")
    echo "scratch /analysis -> HTTP $(curl -s -o /dev/null -w '%{http_code}' "http://127.0.0.1:$PORT/analysis")"
    stop_server
    if grep -rl all_pending "$SCRATCH/web/dist"; then echo "all_pending IN scratch build"; exit 1; else echo "scratch web/dist has no all_pending"; fi
    start_server "$REPO" "$W/smoke-after" "$W/smoke-facts-after.json"
    PORT=$(python3 -c "import json;print(json.load(open('$W/smoke-facts-after.json'))['url'].rsplit(':',1)[1])")
    echo "head /analysis -> HTTP $(curl -s -o /dev/null -w '%{http_code}' "http://127.0.0.1:$PORT/analysis")"
    stop_server
    grep -rl all_pending "$REPO/web/dist" > /dev/null && echo "head web/dist contains all_pending"
    echo "uv: $(uv --version)"; echo "node: $(node --version)"
    echo "@fontsource/plus-jakarta-sans: $(node -p "require('$REPO/web/node_modules/@fontsource/plus-jakarta-sans/package.json').version")"
  } 2>&1 | tee "$W/smoke.tmp"
  [ "${PIPESTATUS[0]}" = 0 ] || exit 1
  mv "$W/smoke.tmp" "$W/smoke.ok"
fi

# ---- Step 3: capture both sides, one after the other
capture() { # $1 = side, $2 = tree
  local side=$1 tree=$2 out=$W/$1
  done_ "$side.captured" && return 0
  rm -rf "$out" "$W/facts-$side.json" "$W/$side-sample"
  start_server "$tree" "$W/$side-sample" "$W/facts-$side.json"
  PORT=$(python3 -c "import json;print(json.load(open('$W/facts-$side.json'))['url'].rsplit(':',1)[1])")
  mkdir -p "$out"
  local t0 t1 rc=0
  t0=$(now)
  (cd "$tree" && node scripts/design/comparisons.mjs --facts "$W/facts-$side.json" --out "$out" --strict) > "$W/$side.console" 2>&1 || rc=$?
  t1=$(now)
  stop_server
  printf 'start: %s\nend: %s\nstrict-exit: %s\n' "$t0" "$t1" "$rc" > "$out/exit-code"
  touch "$W/$side.captured"
}
capture after "$REPO"
capture before "$SCRATCH"

# ---- Step 4: compare (marker: ae-full.txt)
if ! done_ ae-full.txt; then
  rc=0
  node scripts/design/ae.mjs "$W/before" "$W/after" > "$W/ae-full.tmp" 2>&1 || rc=$?
  echo "ae-exit: $rc" > "$W/ae-exit"
  mv "$W/ae-full.tmp" "$W/ae-full.txt"
fi
echo "ALL STEPS DONE for $HEAD_SHA in $W"
```
